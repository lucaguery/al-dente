- **Open question A4 (grid responsive breakpoints) resolution:** 2-col mobile (default) → 3-col @md (`md:grid-cols-3`). Skip 4-col @lg (PWA on iPhone is the primary use case; desktop is rare).
- **Anti-flash hydration pattern (RESEARCH Pattern 7):** SSR initializes `view = "grid"`; `useEffect` reads `localStorage.getItem("aldente.library.view")` post-mount, updates state, marks `hydrated = true`. The panel container has `className={\`transition-opacity duration-150 ${hydrated ? "opacity-100" : "opacity-0"}\`}`. When localStorage value is `"grid"` (the default), there's no visual swap; opacity flips from 0 to 1 immediately. When stored value is `"list"` or `"patina"`, the panel swaps then fades in — total flash time bounded by `requestAnimationFrame + 150ms transition`.
- **localStorage SSR safety:** All localStorage reads in `useEffect` only (RESEARCH Pitfall 5). Never read at module level or in render body.
- **`dogear` defaulting:** `<LedgerCard>` already defaults `dogear` to `patina >= 3` (per 32-02 Task 1 — `const showDogear = dogear ?? patina >= 3;`). Per UI-SPEC §7.1, no explicit `dogear` prop needed in consumers unless the planner wants to force-show or force-hide.
- **List view marginalia (Open question 3 resolution):** Per UI-SPEC §9.2 + Open Question 3 — Phase 32 default: **OMIT** list-view marginalia for v0.7. The doc mock shows "cèpes secs, magique" but there's no field on Recipe model that maps to this. Adding a separate per-recipe `cooking_logs` fetch for every row in list view is expensive and out of scope for this plan. List view shows: title + cuisine · relative date caption + photo. The marginalia comes back in 32-05 (recipe detail) where the data path is simpler.
- **Patine view empty-bucket handling (UI-SPEC §6.3):** Sections with 0 recipes are omitted entirely (no "0 fois" header rendered for an empty bucket).
- **Section header copy:** `home.library.patina_section.heritage` = "Héritage", `.habitudes` = "Habitudes", `.essai` = "À l'essai". Plus a Caveat count suffix (e.g., "Héritage · 3"). The doc mock shows the count in Caveat next to the section label.
- **View-switcher icons:** lucide-react `LayoutGrid`, `List`, `Layers` (per UI-SPEC §1).
- **Anti-flash trick:** Initialize `hydrated` to `false` on mount; flip to `true` inside the `useEffect`. The 150ms transition wraps the panel itself. The view-switcher buttons can be visible immediately (their state matches the SSR default until hydration completes).
- **The `useLibraryView` hook idea:** UI-SPEC §15.C C-04 mentions a hook. Planner's call whether to extract a custom hook (`frontend/lib/hooks/useLibraryView.ts`) or inline the state in `RecipesPage`. **Recommendation:** Inline in `RecipesPage` — single consumer, ~10 lines of state.

```typescript
// ─── Phase 32 §15.C (SOBER-05) — Patine helpers. ──────────────────────
// Source: CONTEXT D-11 + D-12, cross-verified with design-system.html mock
// (Risotto 34× → 3, jamais → 0).

export type PatinaLevel = 0 | 1 | 2 | 3;

/**
 * Map a recipe's cook_count to a patine intensity used by <LedgerCard>.
 *
 *   0      → 0 (À l'essai — jamais cuisiné)
 *   1-2    → 1 (récent)
 *   3-10   → 2 (Habitudes)
 *   >10    → 3 (Héritage)
 */
export function cookCountToPatina(n: number): PatinaLevel {
  if (n === 0) return 0;
  if (n <= 2) return 1;
  if (n <= 10) return 2;
  return 3;
}

/**
 * Bucket recipes into the three Bibliothèque patine sections.
 * Empty buckets are still represented with empty arrays — the page omits
 * the section header when bucket.length === 0 (UI-SPEC §6.3).
 */
export function groupByPatina(recipes: readonly Recipe[]): {
  heritage: Recipe[];
  habitudes: Recipe[];
  essai: Recipe[];
} {
  return {
    heritage: recipes.filter((r) => cookCountToPatina(r.cook_count) >= 3),
    habitudes: recipes.filter((r) => cookCountToPatina(r.cook_count) === 2),
    essai: recipes.filter((r) => cookCountToPatina(r.cook_count) <= 1),
  };
}
```

```tsx
"use client";

// Phase 32 §15.C (SOBER-03) — segmented control for the 3 Bibliothèque views.
// Per CONTEXT D-09 + D-10 + UI-SPEC §9.2.
// State persistence (localStorage["aldente.library.view"]) is owned by the
// PARENT page — this component is a pure controlled segmented input.

import { useTranslations } from "next-intl";
import { LayoutGrid, List, Layers } from "lucide-react";

export type LibraryView = "grid" | "list" | "patina";

export interface LibraryViewSwitchProps {
  value: LibraryView;
  onChange: (next: LibraryView) => void;
  className?: string;
}

const VIEWS: { key: LibraryView; Icon: typeof LayoutGrid }[] = [
  { key: "grid", Icon: LayoutGrid },
  { key: "list", Icon: List },
  { key: "patina", Icon: Layers },
];

export function LibraryViewSwitch({ value, onChange, className }: LibraryViewSwitchProps) {
  const tAria = useTranslations("home.library.view");
  return (
    <div
      role="radiogroup"
      aria-label="Vue de la bibliothèque"
      className={`inline-flex items-center bg-secondary rounded-full ${className ?? ""}`}
      style={{ padding: "3px", gap: "2px" }}
    >
      {VIEWS.map(({ key, Icon }) => {
        const active = value === key;
        return (
          <button
            key={key}
            type="button"
            role="radio"
            aria-checked={active}
            aria-label={tAria(`${key}.aria`)}
            onClick={() => onChange(key)}
            className="inline-flex items-center justify-center rounded-full transition-all duration-150"
            style={{
              padding: "5px 9px",
              background: active ? "var(--card)" : "transparent",
              color: active ? "var(--primary)" : "var(--foreground-muted)",
              boxShadow: active ? "var(--shadow-card)" : "none",
              fontSize: "12px",
              fontWeight: 500,
            }}
          >
            <Icon size={14} aria-hidden />
          </button>
        );
      })}
    </div>
  );
}

export default LibraryViewSwitch;
```

```tsx
export function RecipeRow({ recipe }: RecipeRowProps) {
  const t = useTranslations("recipes");

  // Photo URL resolution — mirror RecipeCard pattern exactly.
```
