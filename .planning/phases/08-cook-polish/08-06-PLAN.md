"use client";

// Phase 8 — Cooking-log history card (COOK-10).
// Per 08-UI-SPEC §"Surface 6": paper-grain Card frame + aspect-[4/3] top photo
// (signed URL via getCookingLogSignedPhotoUrl) + Fraunces text-title recipe
// name (line-clamp-2) + IBM Plex Sans cooked-on date + 3-state rating pill via
// inline ratingChipClass helper. Tap navigates to /recipes/{recipe_id}.
//
// The CookingLogResponse type from lib/cooking.ts is extended locally with a
// `recipe_title` field — required for rendering the editorial title without
// an extra recipe-fetch round-trip. Backend wiring per 08-UI-SPEC §"Phase 8
// budget reality" — the field must be added server-side; if absent, the
// history route's EmptyState fallback fires before any card renders.

import Link from "next/link";
import { useEffect, useState } from "react";
import { useTranslations } from "next-intl";
import {
  getCookingLogSignedPhotoUrl,
  type CookingLogResponse,
  type LogRating,
} from "@/lib/cooking";

export type CookingLogHistoryItem = CookingLogResponse & {
  recipe_title: string;
};

// Per 08-UI-SPEC §"Vote-chip / rating-chip extraction" — inline 3-state helper.
// Phase 7's chipClass operates on VoteState (5-state); LogRating is 3-state and
// not directly compatible. Inline avoids a shared-helper refactor; refactor
// later if a second consumer (e.g. recipe-detail recent ratings) emerges.
function ratingChipClass(rating: LogRating): string {
  const base =
    "inline-flex items-center rounded-full px-2 py-1 text-sm font-medium h-8";
  if (rating === "loved")
    return `${base} bg-surface-rose-100 text-primary border border-primary/40`;
  if (rating === "liked")
    return `${base} bg-[var(--color-valide-tint)] text-foreground border border-emerald-500/30`;
  // disliked
  return `${base} bg-muted text-muted-foreground border border-border`;
}

export function CookingLogCard({ log }: { log: CookingLogHistoryItem }) {
  const tRating = useTranslations("cooking_log.rating");
  const firstPath = log.photo_paths[0] ?? "";
  const [src, setSrc] = useState<string | null>(null);

  useEffect(() => {
    if (!firstPath) return;
    let alive = true;
    getCookingLogSignedPhotoUrl(log.id, firstPath)
      .then((url) => {
        if (alive) setSrc(url);
      })
      .catch(() => {
        // Silent fallback — placeholder div renders via the `src === null` branch.
      });
    return () => {
      alive = false;
    };
  }, [log.id, firstPath]);

  // French relative format for "il y a 2 jours" or "vendredi 8 mai" idiom.
  // Use Intl.DateTimeFormat for absolute fallback when relative would be too
  // far in the past — couple-scale assumes most logs are within ~2 weeks.
  const cookedOnDate = new Date(log.cooked_at);
  const cookedOnLabel = new Intl.DateTimeFormat("fr-FR", {
    weekday: "long",
    day: "numeric",
    month: "long",
  }).format(cookedOnDate);

  return (
    <Link
      href={`/recipes/${log.recipe_id}`}
      className="paper-grain flex flex-col gap-3 p-4 bg-card rounded-xl border border-border shadow-card hover:shadow-card-hover active:translate-y-px transition-all duration-150"
    >
      {src ? (
        // eslint-disable-next-line @next/next/no-img-element -- signed URL is short-lived
        <img
          src={src}
          alt=""
          className="aspect-[4/3] w-full rounded-lg object-cover"
        />
      ) : log.photo_paths.length > 0 ? (
        <div
          aria-hidden
          className="aspect-[4/3] w-full rounded-lg bg-surface-muted"
        />
      ) : null}
      <div className="flex flex-col gap-2">
        <h3 className="text-title line-clamp-2">{log.recipe_title}</h3>
        <div className="flex items-center justify-between gap-3">
          <span className="text-sm text-foreground-muted">{cookedOnLabel}</span>
          {log.rating && (
            <span className={ratingChipClass(log.rating)}>
              {tRating(log.rating)}
            </span>
          )}
        </div>
        {log.notes && (
          <p className="text-sm text-foreground line-clamp-2 leading-relaxed">
            {log.notes}
          </p>
        )}
      </div>
    </Link>
  );
}
```

"use client";

// Phase 8 — Cooking-log history view (COOK-10). "What we ate this week."
//
// Frontend-only deliverable. The route fetches GET /api/cooking-logs?days=14
// (best-effort — endpoint may not yet exist). On success, logs are grouped by
// the absolute French date label of `cooked_at`, then rendered as
// CookingLogCard rows inside dated <section> blocks. On failure (404 or any
// other error), the EmptyState component renders with placeholder copy
// reusing recipes.empty_heading / _body per 08-UI-SPEC §"Copywriting Contract"
// (TODO(productize) — cooking-log-specific copy).
//
// Per 08-UI-SPEC §"Surface 6" Path B: the sticky-header heading is omitted
// (BottomNav links here directly; no back-arrow surface; the first dated
// section header IS the page anchor).

import { useEffect, useState } from "react";
import { useTranslations } from "next-intl";
import { ChefHat } from "lucide-react";
import { EmptyState } from "@/components/EmptyState";
import {
  CookingLogCard,
  type CookingLogHistoryItem,
} from "@/components/CookingLogCard";
import { OnboardingGuard } from "@/lib/onboarding-guard";
import { api } from "@/lib/api";

type LoadState =
  | { kind: "loading" }
  | { kind: "ready"; logs: CookingLogHistoryItem[] }
  | { kind: "empty" };

function groupByDateFr(
  logs: CookingLogHistoryItem[],
): Array<[string, CookingLogHistoryItem[]]> {
  const fmt = new Intl.DateTimeFormat("fr-FR", {
    weekday: "long",
    day: "numeric",
    month: "long",
  });
  const groups = new Map<string, CookingLogHistoryItem[]>();
  for (const log of logs) {
    const label = fmt.format(new Date(log.cooked_at));
    const bucket = groups.get(label) ?? [];
    bucket.push(log);
    groups.set(label, bucket);
  }
  // Preserve insertion order; assumes the backend returns logs in
  // descending cooked_at — most-recent-first.
  return Array.from(groups.entries());
}

export default function CookingLogsHistoryPage() {
  const t = useTranslations("recipes"); // placeholder: empty_heading / empty_body reused per 08-UI-SPEC TODO(productize)
  const [state, setState] = useState<LoadState>({ kind: "loading" });

  useEffect(() => {
    let alive = true;
    (async () => {
      try {
        const logs = await api<CookingLogHistoryItem[]>(
          "/api/cooking-logs?days=14",
        );
        if (!alive) return;
        if (logs.length === 0) {
          setState({ kind: "empty" });
        } else {
          setState({ kind: "ready", logs });
        }
      } catch {
        // 404 (endpoint not yet shipped) OR any other failure — fall back to
        // EmptyState rather than a stale error UI.
        if (alive) setState({ kind: "empty" });
      }
    })();
    return () => {
      alive = false;
    };
  }, []);

  return (
    <OnboardingGuard>
      <section className="flex flex-col flex-1 bg-background">
        <div className="px-6 pt-8 pb-24 flex flex-col gap-6">
          {state.kind === "loading" ? (
            <div className="flex flex-col gap-3">
              <div className="h-8 w-1/2 rounded bg-surface-muted animate-pulse" />
              <div className="h-64 w-full rounded-xl bg-surface-muted animate-pulse" />
              <div className="h-64 w-full rounded-xl bg-surface-muted animate-pulse" />
            </div>
          ) : state.kind === "empty" ? (
            <EmptyState
              icon={ChefHat}
              heading={t("empty_heading")}
              body={t("empty_body")}
            />
          ) : (
            groupByDateFr(state.logs).map(([dateLabel, logsInGroup]) => (
              <section key={dateLabel} className="flex flex-col gap-3">
                <h2 className="font-display italic text-base text-foreground pt-6 pb-2">
                  {dateLabel}
                </h2>
                {logsInGroup.map((log) => (
                  <CookingLogCard key={log.id} log={log} />
                ))}
              </section>
            ))
          )}
        </div>
      </section>
    </OnboardingGuard>
  );
}
```

**Key implementation choices:**

1. **No sticky header heading** (08-UI-SPEC §"Surface 6" Path B locked decision): the BottomNav links directly to /cooking-logs; no back-arrow needed; the first dated section header IS the page anchor. **No new i18n keys for a page heading.**

2. **EmptyState placeholder copy reuse** (08-UI-SPEC §"Copywriting Contract" + §"Out of Scope"): uses `t("empty_heading")` / `t("empty_body")` from the `recipes` namespace as placeholder reuse.
