    **Three pure Gemini-call functions** (no DB, no broadcast — caller wraps these):

    ```python
    _EXTRACT_PROMPT_VOICE = (
        "Extrais les champs structurés de cette recette dictée en français. "
        "Renvoie null pour les champs absents — n'invente rien. Ne mets que des valeurs "
        "des vocabulaires verrouillés pour cuisine, mood, main_protein, seasonality."
    )
    _EXTRACT_PROMPT_PHOTOS = (
        "Voici une recette photographiée (1 à 4 images). Extrais les champs structurés "
        "en français. Renvoie null pour les champs absents — n'invente rien."
    )
    _MODIFY_PROMPT = (
        "Voici une recette existante (JSON) et une instruction de modification dictée en "
        "français. Renvoie la recette MODIFIÉE en respectant le même schéma. "
        "Conserve les champs non concernés tels quels."
    )


    def extract_from_transcript(transcript: str) -> GeminiExtractedRecipe:
        """Voice transcript -> structured recipe. Raises on Gemini error."""
        response = _gemini().models.generate_content(
            model="gemini-2.5-flash",
            contents=[_EXTRACT_PROMPT_VOICE, transcript],
            config=types.GenerateContentConfig(
                response_mime_type="application/json",
                response_schema=GeminiExtractedRecipe,
            ),
        )
        parsed = response.parsed
        if not isinstance(parsed, GeminiExtractedRecipe):
            raise ValueError("Gemini did not return a valid GeminiExtractedRecipe")
        return parsed


    def extract_from_photos(photo_bytes_list: list[bytes]) -> GeminiExtractedRecipe:
        """1-4 photos -> structured recipe. Inline bytes path; <20 MB total."""
        if not photo_bytes_list:
            raise ValueError("at least one photo required")
        if len(photo_bytes_list) > 4:
            raise ValueError("at most 4 photos accepted")
        # mime_type: pass image/jpeg for all — Gemini auto-detects from magic bytes.
        # The router has already sniffed/validated MIME via storage.detect_mime_and_ext.
        parts = [
            types.Part.from_bytes(data=b, mime_type="image/jpeg")
            for b in photo_bytes_list
        ]
        response = _gemini().models.generate_content(
            model="gemini-2.5-flash",
            contents=[_EXTRACT_PROMPT_PHOTOS, *parts],
            config=types.GenerateContentConfig(
                response_mime_type="application/json",
                response_schema=GeminiExtractedRecipe,
            ),
        )
        parsed = response.parsed
        if not isinstance(parsed, GeminiExtractedRecipe):
            raise ValueError("Gemini did not return a valid GeminiExtractedRecipe")
        return parsed


    def apply_voice_modification(
        recipe_json: dict, transcript: str
    ) -> GeminiExtractedRecipe:
        """Existing recipe + modification instruction -> modified structured recipe.
        Read-only — does NOT persist. Caller (router) returns to the FE for review."""
        import json as _json
        response = _gemini().models.generate_content(
            model="gemini-2.5-flash",
            contents=[
                _MODIFY_PROMPT,
                f"Recette actuelle: {_json.dumps(recipe_json, ensure_ascii=False)}",
                f"Instruction: {transcript}",
            ],
            config=types.GenerateContentConfig(
                response_mime_type="application/json",
                response_schema=GeminiExtractedRecipe,
            ),
        )
        parsed = response.parsed
        if not isinstance(parsed, GeminiExtractedRecipe):
            raise ValueError("Gemini did not return a valid GeminiExtractedRecipe")
        return parsed
    ```

    **BackgroundTask bodies** — three sync functions that the router queues. Each opens its own SessionLocal, runs its Gemini call, persists the result, and broadcasts on success. On exception: writes promotion_error, increments promotion_attempts, leaves status='draft'. NEVER raises.

    ```python
    def _apply_extracted(recipe: Recipe, extracted: GeminiExtractedRecipe) -> None:
        """Apply Gemini's parsed output to the recipe row in-place."""
        if not extracted.title or not extracted.title.strip():
            raise ValueError("Gemini returned empty title")
        recipe.title = extracted.title
        recipe.ingredients = (
            [i.model_dump() for i in extracted.ingredients]
            if extracted.ingredients else None
        )
        recipe.steps = extracted.steps
        recipe.prep_time_minutes = extracted.prep_time_minutes
        recipe.servings = extracted.servings
        recipe.cuisine = extracted.cuisine
        recipe.mood = list(extracted.mood) if extracted.mood else []
        recipe.main_protein = extracted.main_protein
        recipe.seasonality = (
            list(extracted.seasonality) if extracted.seasonality
            else ["spring", "summer", "autumn", "winter"]
        )
        recipe.status = "structured"
        recipe.promotion_error = None


    def _broadcast_promoted(recipe: Recipe) -> None:
        payload = RecipeResponse.model_validate(recipe).model_dump(mode="json")
        # broadcast_to_household is async; the BackgroundTask runs sync.
        asyncio.run(
            broadcast_to_household(recipe.household_id, "recipe.promoted", payload)
        )


    def _record_failure(db, recipe: Recipe, exc: Exception) -> None:
        log.exception("promotion failed recipe=%s", recipe.id)
        recipe.promotion_error = str(exc)[:500]
        recipe.promotion_attempts = (recipe.promotion_attempts or 0) + 1
        db.commit()


    def promote_voice_draft(recipe_id: UUID, transcript: str) -> None:
        """BackgroundTask body for POST /recipes/voice."""
        db = SessionLocal()
        try:
            recipe = db.scalar(select(Recipe).where(Recipe.id == recipe_id))
            if recipe is None:
                log.warning("promote_voice: recipe %s vanished", recipe_id)
                return
            try:
                extracted = extract_from_transcript(transcript)
                _apply_extracted(recipe, extracted)
                recipe.promotion_attempts = (recipe.promotion_attempts or 0) + 1
                db.commit()
                db.refresh(recipe)
                _broadcast_promoted(recipe)
            except Exception as exc:  # noqa: BLE001 — must never raise out of task
                _record_failure(db, recipe, exc)
        finally:
            db.close()


    def promote_photo_draft(recipe_id: UUID, photo_bytes_list: list[bytes]) -> None:
        """BackgroundTask body for POST /recipes/photo."""
        db = SessionLocal()
        try:
            recipe = db.scalar(select(Recipe).where(Recipe.id == recipe_id))
            if recipe is None:
                log.warning("promote_photo: recipe %s vanished", recipe_id)
                return
            try:
                extracted = extract_from_photos(photo_bytes_list)
                _apply_extracted(recipe, extracted)
                recipe.promotion_attempts = (recipe.promotion_attempts or 0) + 1
                db.commit()
                db.refresh(recipe)
                _broadcast_promoted(recipe)
            except Exception as exc:  # noqa: BLE001
                _record_failure(db, recipe, exc)
        finally:
            db.close()


    def retry_promotion(recipe_id: UUID) -> None:
        """BackgroundTask body for POST /recipes/{id}/retry-promotion.

        Re-reads source_capture to get the original transcript / photo paths,
        clears promotion_error, then dispatches to the appropriate promote_*.
        """
        db = SessionLocal()
        try:
            recipe = db.scalar(select(Recipe).where(Recipe.id == recipe_id))
            if recipe is None:
                return
            sc = recipe.source_capture or {}
            sc_type = sc.get("type")
            payload = sc.get("payload") or {}
            recipe.promotion_error = None
            db.commit()

            if sc_type == "voice":
                transcript = payload.get("transcript") or ""
                if not transcript.strip():
                    _record_failure(
                        db, recipe, ValueError("retry: transcript missing"),
                    )
                    return
                # Drop the request session — promote_voice_draft opens its own.
                db.close()
                promote_voice_draft(recipe_id, transcript)
                return
            if sc_type == "photo":
                # photo bytes aren't stored in source_capture (only paths). Re-download
                # from Supabase Storage in a follow-up if dogfood needs it; in v0.1
                # photo retries record a clear error to surface productize-later work.
                _record_failure(
                    db, recipe,
                    ValueError(
                        "photo retry not supported in v0.1 — # TODO(productize)"
                    ),
                )
                return
            # url / manual / unknown — should never reach retry path
            _record_failure(
                db, recipe, ValueError(f"retry not applicable for type={sc_type!r}")
            )
        finally:
            db.close()
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Frontend → backend | Voice transcripts and photo bytes cross via authenticated POST (cookie auth). Untrusted natural-language input. |
| Backend → Gemini API | Outbound to Google. Service key (`GEMINI_API_KEY`) leaves the boundary; never returned to the client. |

| BackgroundTask → DB | Internal; same trust as the request handler that queued it. |
