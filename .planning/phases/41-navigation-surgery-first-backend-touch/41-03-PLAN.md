<objective>
Replace `/recipes/new` (currently mounts `<RecipeThread mode="capture" />` directly) with a 5-option numbered picker. Move the thread-capture mount to a new dynamic route `/recipes/new/[surface]` that pre-seeds the composer per surface. Note rapide bypasses the thread entirely with a name-only modal that POSTs `{name}` to `/recipes` and redirects to the structured view.

Purpose: The current `/recipes/new` plunges the user straight into a conversational thread even when they just want to scribble a name to come back to later. Per sketch §Ajouter (lines 1714-1755), recipe creation is a one-tap-then-thread flow: pick the method, then enter the surface. D-09 + D-11 (in-thread unification) is preserved — the chooser is upstream of the thread; inside `/recipes/new/[surface]` there are no tabs. D-01 (stateless) means no resume-in-flight UI noise — every tap creates a fresh draft.

Output: Two new route files, two new components (`NouvellePicker`, `NoteRapideModal`), several new i18n keys, and one new Playwright spec. The existing `<RecipeThread mode="capture" />` component is unchanged — only its host route moves.
</objective>

  ```
  'use client';
  import { useState, useTransition } from 'react';
  import { useRouter } from 'next/navigation';
  import { useTranslations } from 'next-intl';
  import { toast } from 'sonner';
  import { Dialog, DialogContent, DialogTitle, DialogDescription } from '@/components/ui/dialog'; // adjust to project's primitive
  import { Button } from '@/components/ui/button';
  import { Input } from '@/components/ui/input';
  import { api } from '@/lib/api';

  type Props = { open: boolean; onOpenChange: (v: boolean) => void };

  export function NoteRapideModal({ open, onOpenChange }: Props) {
    const router = useRouter();
    const t = useTranslations('recipes.new.note_rapide');
    const [name, setName] = useState('');
    const [isPending, startTransition] = useTransition();

    const onSubmit = async (e: React.FormEvent) => {
      e.preventDefault();
      const trimmed = name.trim();
      if (!trimmed) return;
      startTransition(async () => {
        try {
          const recipe = await api<{ id: string }>('/recipes', {
            method: 'POST',
            body: JSON.stringify({ name: trimmed }),
          });
          onOpenChange(false);
          router.push(`/recipes/${recipe.id}`); // structured view per D-02
        } catch (err) {
          toast.error(t('error'));
        }
      });
    };

    return (
      <Dialog open={open} onOpenChange={onOpenChange}>
        <DialogContent>
          <DialogTitle>{t('title')}</DialogTitle>
          <DialogDescription className="sr-only">{t('description')}</DialogDescription>
          <form onSubmit={onSubmit} className="flex flex-col gap-4 mt-4">
            <Input
              autoFocus
              value={name}
              onChange={(e) => setName(e.target.value)}
              placeholder={t('placeholder')}
              maxLength={80}
            />
            <Button type="submit" disabled={isPending || !name.trim()}>
              {t('cta')}
            </Button>
          </form>
        </DialogContent>
      </Dialog>
    );
  }
  ```
