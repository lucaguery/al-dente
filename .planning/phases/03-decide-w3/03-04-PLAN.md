    ```tsx
    "use client";

    // Phase 3 — shortlist swipe deck. Renders up to 2 visible cards (front + peek).
    // Owns: (1) the visible-card index, (2) optimistic vote handling with rollback
    // on POST failure, (3) wires ShortlistCard → postVote → parent's onShortlistUpdate.
    //
    // 03-UI-SPEC.md §Surface 5 + §"Interaction Patterns > Optimistic UI"

    import { useState } from "react";
    import { AnimatePresence } from "framer-motion";
    import { useTranslations } from "next-intl";
    import { toast } from "sonner";
    import {
      ShortlistCard,
      ShortlistThumbButtons,
    } from "@/components/ShortlistCard";
    import { postVote, type ShortlistVote, type VoteValue } from "@/lib/votes";
    import type { Recipe } from "@/lib/recipes";

    export type DeckMember = {
      id: string;
      name: string;
      color_hex: string;
    };

    export type ShortlistDeckProps = {
      shortlistId: string;
      recipes: Recipe[];
      votes: ShortlistVote[];
      me: DeckMember;
      partner: DeckMember;
      onVoteApplied: (vote: ShortlistVote) => void;
    };

    function partnerVoteFor(
      votes: ShortlistVote[],
      recipeId: string,
      partnerId: string,
    ): "yes" | "no" | "unvoted" {
      const v = votes.find(
        (x) => x.recipe_id === recipeId && x.member_id === partnerId,
      );
      if (!v) return "unvoted";
      return v.vote;
    }

    export function ShortlistDeck({
      shortlistId,
      recipes,
      votes,
      me,
      partner,
      onVoteApplied,
    }: ShortlistDeckProps) {
      const t = useTranslations("home.shortlist");
      const [index, setIndex] = useState(0);
      const [submittingFor, setSubmittingFor] = useState<string | null>(null);

      const remaining = recipes.slice(index);
      if (remaining.length === 0) {
        // Caller renders VoteSummary in this case; this component handles only the deck phase.
        return null;
      }
      const current = remaining[0];
      const next = remaining[1];

      async function handleVote(value: VoteValue) {
        if (submittingFor) return;
        const recipeId = current.id;
        setSubmittingFor(recipeId);
        // Optimistic: advance the deck immediately. The parent receives the vote
        // via onVoteApplied and updates its votes[] state for the summary view.
        const optimistic: ShortlistVote = {
          shortlist_id: shortlistId,
          recipe_id: recipeId,
          member_id: me.id,
          vote: value,
        };
        onVoteApplied(optimistic);
        const previousIndex = index;
        setIndex((i) => i + 1);
        try {
          await postVote(shortlistId, recipeId, value);
        } catch (err) {
          // Rollback: revert deck position and show toast. The caller is responsible for
          // removing the optimistic vote from votes[] if it cares (most realistic flow:
          // the next vote.created event from the server will reconcile).
          setIndex(previousIndex);
          toast.error(t("vote_failed"));
        } finally {
          setSubmittingFor(null);
        }
      }

      return (
        <div className="flex flex-col flex-1 items-center justify-center px-4 pt-4 pb-8 gap-6">
          <div className="relative w-full max-w-sm aspect-[3/4]">
            {next && (
              <ShortlistCard
                key={`peek-${next.id}`}
                recipe={next}
                partnerVote={partnerVoteFor(votes, next.id, partner.id)}
                partnerName={partner.name}
                partnerColorHex={partner.color_hex}
                onVote={() => {}}
                isFront={false}
              />
            )}
            <AnimatePresence mode="wait">
              <ShortlistCard
                key={`front-${current.id}`}
                recipe={current}
                partnerVote={partnerVoteFor(votes, current.id, partner.id)}
                partnerName={partner.name}
                partnerColorHex={partner.color_hex}
    ```

    ```tsx
    // Phase 3 — top-level Home content router. Replaces the hero+CTA block in
    // app/page.tsx. Owns: shortlist + active-cooking-log fetch, realtime updates,
    // CTA orchestration.
    //
    // 03-UI-SPEC.md §"Home tab content tree".

    import { useCallback, useEffect, useMemo, useState } from "react";
    import { Sparkles } from "lucide-react";
    import { useTranslations } from "next-intl";
    import { toast } from "sonner";
    import Link from "next/link";
    import { Button } from "@/components/ui/button";
    import { EmptyState } from "@/components/EmptyState";
    import { ShortlistDeck } from "@/components/ShortlistDeck";
    import { VoteSummary } from "@/components/VoteSummary";
    import { CookingBanner } from "@/components/CookingBanner";
    import { ColdStartChip } from "@/components/ColdStartChip";
    import { RegenerateSheet } from "@/components/RegenerateSheet";
    import { useSession } from "@/components/SessionProvider";
    import {
      VOTE_CREATED_DOM_EVENT,
      SHORTLIST_CREATED_DOM_EVENT,
      COOKING_STARTED_DOM_EVENT,
      type Phase3VoteEvent,
      type Phase3CookingStartedEvent,
    } from "@/components/RealtimeProvider";
    import {
      computeVoteState,
      delegateShortlist,
      type ShortlistVote,
    } from "@/lib/votes";
    import {
      fetchTodayShortlist,
      regenerateShortlist,
      type ShortlistFilters,
      type ShortlistResponse,
    } from "@/lib/shortlist";
    import {
      getActiveCookingLog,
      postStartCooking,
      type CookingLogResponse,
    } from "@/lib/cooking";

    const COOK_BANNER_SKIP_KEY = "dismissed_cooking_banner_at";

    type Member = { id: string; name: string; color_hex: string };

    export function HomeDecide() {
      const t = useTranslations("home");
      const tShortlist = useTranslations("home.shortlist");
      const tSummary = useTranslations("home.summary");
      const session = useSession();

      const [shortlist, setShortlist] = useState<ShortlistResponse | null>(null);
      const [shortlistLoaded, setShortlistLoaded] = useState(false);
      const [activeLog, setActiveLog] = useState<CookingLogResponse | null>(null);
      const [bannerSkipped, setBannerSkipped] = useState(false);
      const [regenOpen, setRegenOpen] = useState(false);
      const [regenSubmitting, setRegenSubmitting] = useState(false);
      const [delegateInFlight, setDelegateInFlight] = useState(false);
      const [cookInFlight, setCookInFlight] = useState(false);
      const [validéToastedFor] = useState<Set<string>>(() => new Set());

      // ── Resolve me + partner from session.household.members ────────────────
      const me: Member | null = useMemo(() => {
        const member = (session as { member?: Member }).member;
        return member ?? null;
      }, [session]);
      const partner: Member | null = useMemo(() => {
        const household = (
          session as {
            household?: { members?: Member[] };
          }
        ).household;
        const members = household?.members ?? [];
        if (!me) return null;
        const other = members.find((m) => m.id !== me.id);
        return other ?? null;
      }, [session, me]);

      // ── Initial fetch ──────────────────────────────────────────────────────
      useEffect(() => {
        let cancelled = false;
        Promise.all([fetchTodayShortlist(), getActiveCookingLog()])
          .then(([sl, log]) => {
            if (cancelled) return;
            setShortlist(sl);
            setShortlistLoaded(true);
            setActiveLog(log);
            // Restore session-skipped banner state
            try {
              if (window.sessionStorage.getItem(COOK_BANNER_SKIP_KEY)) {
                setBannerSkipped(true);
              }
            } catch {
              /* ignore */
            }
          })
          .catch(() => {
            if (cancelled) return;
            setShortlistLoaded(true);
          });
        return () => {
          cancelled = true;
        };
      }, []);

      // ── Realtime: vote.created — partner or self echo ──────────────────────
      useEffect(() => {
        function onVoteCreated(e: Event) {
          const payload = (e as CustomEvent<Phase3VoteEvent>).detail;
          if (!shortlist || payload.shortlist_id !== shortlist.shortlist_id) {
            return;
          }
          // Reconcile votes[] — replace any existing row for (recipe, member) or append.
          setShortlist((prev) => {
            if (!prev) return prev;
            const others = prev.votes.filter(
              (v) =>
                !(
                  v.recipe_id === payload.recipe_id &&
                  v.member_id === payload.member_id
                ),
            );
            const next: ShortlistVote = {
              shortlist_id: payload.shortlist_id,
              recipe_id: payload.recipe_id,
              member_id: payload.member_id,
              vote: payload.vote,
            };
            return { ...prev, votes: [...others, next] };
          });

          // Drift detection (architecture invariant #2 sanity)
          if (me && partner) {
            const recipeVotes = [
              ...shortlist.votes.filter(
                (v) =>
                  v.recipe_id === payload.recipe_id &&
                  v.member_id !== payload.member_id,
              ),
              {
                shortlist_id: payload.shortlist_id,
                recipe_id: payload.recipe_id,
                member_id: payload.member_id,
                vote: payload.vote,
              },
            ];
            const local = computeVoteState(recipeVotes, 2);
            if (local !== payload.state) {
              // eslint-disable-next-line no-console
              console.warn(
                "vote-state drift: local=%s server=%s",
                local,
                payload.state,
              );
            }
          }

          // Pressenti → Validé celebration toast (rate-limited per recipe)
          if (
            me &&
            payload.state === "valide" &&
            payload.member_id !== me.id &&
            !validéToastedFor.has(payload.recipe_id)
          ) {
            const recipe = shortlist.recipes.find(
              (r) => r.id === payload.recipe_id,
            );
            if (recipe) {
              validéToastedFor.add(payload.recipe_id);
              toast.success(
                tShortlist("toast_validé", { title: recipe.title }),
                { id: `valide-${payload.recipe_id}` },
              );
            }
          }
        }
        window.addEventListener(VOTE_CREATED_DOM_EVENT, onVoteCreated);
        return () =>
          window.removeEventListener(VOTE_CREATED_DOM_EVENT, onVoteCreated);
      }, [shortlist, me, partner, tShortlist, validéToastedFor]);

      // ── Realtime: shortlist.created — refetch ──────────────────────────────
      useEffect(() => {
        function onShortlistCreated() {
          fetchTodayShortlist().then((sl) => {
            setShortlist(sl);
            if (sl) {
              toast(tShortlist("toast_arrived"));
            }
          });
        }
        window.addEventListener(
          SHORTLIST_CREATED_DOM_EVENT,
          onShortlistCreated,
        );
        return () =>
          window.removeEventListener(
            SHORTLIST_CREATED_DOM_EVENT,
            onShortlistCreated,
          );
      }, [tShortlist]);

      // ── Realtime: cooking.started — partner started cooking ────────────────
      useEffect(() => {
        function onCookingStarted(e: Event) {
          const payload = (e as CustomEvent<Phase3CookingStartedEvent>).detail;
          // Refetch the active log to get the full record (we only have ids in the payload).
          getActiveCookingLog().then(setActiveLog);
        }
        window.addEventListener(COOKING_STARTED_DOM_EVENT, onCookingStarted);
        return () =>
          window.removeEventListener(
            COOKING_STARTED_DOM_EVENT,
            onCookingStarted,
          );
      }, []);

      // ── Optimistic vote handler ───────────────────────────────────────────
      const handleVoteApplied = useCallback((vote: ShortlistVote) => {
        setShortlist((prev) => {
          if (!prev) return prev;
          const others = prev.votes.filter(
            (v) =>
              !(
                v.recipe_id === vote.recipe_id &&
                v.member_id === vote.member_id
              ),
          );
          return { ...prev, votes: [...others, vote] };
        });
      }, []);

      // ── Regenerate flow ───────────────────────────────────────────────────
      const handleRegenerate = useCallback(
        async (filters: ShortlistFilters) => {
          setRegenSubmitting(true);
          try {
            const fresh = await regenerateShortlist(filters);
            setShortlist(fresh);
            setRegenOpen(false);
          } catch {
            toast.error(tSummary("regenerate_failed"));
          } finally {
            setRegenSubmitting(false);
          }
        },
        [tSummary],
      );

      // ── Delegate flow ─────────────────────────────────────────────────────
      const handleDelegate = useCallback(async () => {
        if (!shortlist) return;
        setDelegateInFlight(true);
        try {
          await delegateShortlist(shortlist.shortlist_id);
          toast.success(tSummary("toast_delegated"));
          // The 5 vote.created events from the server will reconcile state.
        } catch {
          toast.error(tSummary("delegate_failed"));
        } finally {
          setDelegateInFlight(false);
        }
      }, [shortlist, tSummary]);

      // ── Cook-start flow ───────────────────────────────────────────────────
      const handleCookStart = useCallback(
        async (recipeId: string) => {
          setCookInFlight(true);
          try {
            const log = await postStartCooking(recipeId);
            setActiveLog(log);
            // Clear the session-skip flag so banner shows
            try {
              window.sessionStorage.removeItem(COOK_BANNER_SKIP_KEY);
            } catch {
              /* ignore */
            }
            setBannerSkipped(false);
            toast.success(tSummary("toast_cooking_started"));
          } catch {
            toast.error(tSummary("cook_failed"));
          } finally {
            setCookInFlight(false);
          }
        },
        [tSummary],
      );

      // ── Banner skip ───────────────────────────────────────────────────────
      const handleBannerSkip = useCallback(() => {
        try {
          window.sessionStorage.setItem(
            COOK_BANNER_SKIP_KEY,
            new Date().toISOString(),
          );
        } catch {
          /* ignore */
        }
        setBannerSkipped(true);
      }, []);

      // ── Render ────────────────────────────────────────────────────────────
      const showCorpusColdStart =
        shortlist !== null && shortlist.recipes.length > 0 && shortlist.recipes.length < 10;

      const cookingBannerVisible =
        activeLog !== null && !bannerSkipped;

      // While the session is loading or shortlist hasn't been fetched, render nothing
      // (the page also renders a loading.tsx skeleton at the segment level if present).
      if (!shortlistLoaded || !me || !partner) {
        return null;
      }

      // No shortlist for today — empty state + cold-start chip if applicable.
      if (shortlist === null) {
        return (
          <div className="flex flex-col flex-1">
            {cookingBannerVisible && activeLog && (
              <CookingBanner
                logId={activeLog.id}
                recipeTitle={recipeTitleFor(activeLog.recipe_id, [])}
                onSkip={handleBannerSkip}
              />
            )}
            <ColdStartChip />
            <EmptyState
              icon={Sparkles}
              heading={tShortlist("empty_heading")}
              body={tShortlist("empty_body")}
              cta={{
                href: "/recipes/new",
                label: tShortlist("empty_cta"),
              }}
            />
          </div>
        );
      }

      // Filter out rejete recipes from the deck queue (D-06 — never appear).
      const memberCount = 2;
      const dealableRecipes = shortlist.recipes.filter((r) => {
        const recipeVotes = shortlist.votes.filter((v) => v.recipe_id === r.id);
        return computeVoteState(recipeVotes, memberCount) !== "rejete";
      });
      // A card is "fully voted by me" if I have a vote on it.
      const myVotes = new Set(
        shortlist.votes
          .filter((v) => v.member_id === me.id)
          .map((v) => v.recipe_id),
      );
      const unvotedByMe = dealableRecipes.filter((r) => !myVotes.has(r.id));
      const allVoted = unvotedByMe.length === 0;

      return (
        <div className="flex flex-col flex-1">
          {cookingBannerVisible && activeLog && (
            <CookingBanner
              logId={activeLog.id}
              recipeTitle={recipeTitleFor(activeLog.recipe_id, shortlist.recipes)}
              onSkip={handleBannerSkip}
            />
          )}
          {showCorpusColdStart && <ColdStartChip />}

          {allVoted ? (
            <VoteSummary
              recipes={dealableRecipes}
              votes={shortlist.votes}
              me={me}
              partner={partner}
              onCookStart={handleCookStart}
              onDelegate={handleDelegate}
              onRegenerate={() => setRegenOpen(true)}
              cookInFlight={cookInFlight}
              delegateInFlight={delegateInFlight}
            />
          ) : (
            <ShortlistDeck
              shortlistId={shortlist.shortlist_id}
              recipes={unvotedByMe}
              votes={shortlist.votes}
              me={me}
              partner={partner}
              onVoteApplied={handleVoteApplied}
            />
          )}

          <RegenerateSheet
            open={regenOpen}
            onOpenChange={setRegenOpen}
            onApply={handleRegenerate}
            submitting={regenSubmitting}
          />
        </div>
      );
    }

    function recipeTitleFor(recipeId: string, recipes: { id: string; title: string }[]): string {
      return recipes.find((r) => r.id === recipeId)?.title ?? "";
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| WS frame → DOM CustomEvent | RealtimeProvider re-fires WS payloads as `window` CustomEvents. Both ends are same-origin. No untrusted source can dispatch — only the RealtimeProvider, which gates on the authenticated WS singleton. |
| Optimistic UI state → backend | The deck advances before the POST resolves; on failure, the index rolls back. The optimistic vote in the local `votes[]` array is reconciled by the inevitable `vote.created` echo (which contains the canonical state). |
| Session storage → CookingBanner skip | sessionStorage is per-origin/per-tab. The skip flag is non-sensitive (boolean-equivalent). |
| useSession output → me/partner derivation | SessionProvider is the single source of truth for "who am I". A spoofed session would already be a fundamental compromise upstream. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-04-01 | Tampering | Optimistic vote could be retained client-side after a backend rejection (e.g., 401) | mitigate | The catch block in `ShortlistDeck::handleVote` rolls back the deck index AND surfaces a toast. The optimistic entry in `shortlist.votes` is overwritten by the next `vote.created` event from the server (the source of truth). 401 specifically is handled by `api()` which clears legacy localStorage and redirects to /onboarding/welcome. |
| T-03-04-02 | Information disclosure | A malicious page in another tab could dispatch the same CustomEvent name to spoof a vote update | accept | CustomEvents are per-document; cross-tab dispatch is not possible. A JS extension running in the same origin could spoof — but at that point the entire page is compromised. Standard browser security model. |
| T-03-04-03 | Information disclosure | `recipeTitleFor` shows recipe title in the banner — could include sensitive content | accept | Recipe titles are user-authored within the household. Visible to all household members by design. |
| T-03-04-04 | Tampering | "Passer" cooking-banner skip persists in sessionStorage — could be cleared by user devtools to re-show banner | accept | Intentional UX: banner is a soft signal. Phase 4 owns full finalization UX. |
| T-03-04-05 | Repudiation | Pressenti → Validé toast has rate-limit by recipe — partner could rapid-toggle vote to spam | accept | Toast id `valide-${recipe.id}` plus the local `validéToastedFor` Set ensure at most ONE toast per recipe per session. Even if a partner toggles repeatedly, the user sees the toast once. |
| T-03-04-06 | Spoofing | `me` and `partner` derived from `useSession()`; if SessionProvider returned a stale value, votes could attribute to wrong member | mitigate | `me` is used ONLY for client-side filtering (`unvotedByMe`) and the optimistic vote's `member_id` field. The backend ALWAYS derives `member_id` from the auth cookie — the optimistic value is overwritten by the server's `vote.created` echo. |
