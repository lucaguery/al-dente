    import pathlib

    photos_dir = pathlib.Path('app/cli/synthetic_photos')
    expected_slugs = [s['slug'] for s in _recipe_specs()]
    assert len(expected_slugs) == 21, f'expected 21 slugs, got {len(expected_slugs)}'

    errors = []
    total_bytes = 0
    for slug in expected_slugs:
        p = photos_dir / f'{slug}.jpg'
        if not p.exists():
            errors.append(f'MISSING: {p}')
            continue
        # Magic bytes
        head = p.read_bytes()[:16]
        sniffed = detect_mime_and_ext(head + b'\\x00' * 16)  # pad in case file is tiny
        if sniffed is None or sniffed[0] != 'image/jpeg':
            errors.append(f'NOT JPEG: {p} (head: {head!r})')
            continue
        # Size sanity
        size = p.stat().st_size
        total_bytes += size
        if size < 10_000:
            errors.append(f'TOO SMALL ({size} bytes — placeholder?): {p}')
        if size > 500_000:
            errors.append(f'TOO LARGE ({size} bytes — re-encode at lower quality): {p}')

    # Filename strictness — no extra files, no wrong extensions
    found = sorted(p.name for p in photos_dir.glob('*'))
    expected = sorted([f'{s}.jpg' for s in expected_slugs] + ['README.md'])
    extra = set(found) - set(expected)
    if extra:

    if errors:
        for e in errors:
