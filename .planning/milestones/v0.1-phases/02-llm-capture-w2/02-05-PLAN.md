        className,
        disabled,

        // Resolve i18n keys at render time. Callers can pass any namespace key

        const tPlaceholder = useTranslationKey(placeholderKey);
        const tAria = useTranslationKey(ariaLabelKey);
        return (
          <Textarea
            value={value}
            onChange={(e) => onChange(e.target.value)}
            placeholder={tPlaceholder}
            aria-label={tAria}
            className={className}
            rows={rows}
            disabled={disabled}
          />
        );
      }

      // Helper: resolve a dotted i18n key like "ns.sub.key" through useTranslations.
      function useTranslationKey(key: string): string {
        const dotIdx = key.lastIndexOf(".");
        const namespace = dotIdx >= 0 ? key.slice(0, dotIdx) : "";
        const leaf = dotIdx >= 0 ? key.slice(dotIdx + 1) : key;
        const t = useTranslations(namespace || undefined);
        return t(leaf);
