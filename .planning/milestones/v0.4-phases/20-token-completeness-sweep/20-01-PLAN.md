Add a helper:
```ts
export function getMemberColorVars(hex: string): { bgVar: string; fgVar: string } {
  const entry = MEMBER_COLORS.find((c) => c.hex === hex);
  return entry
    ? { bgVar: entry.bgVar, fgVar: entry.fgVar }
    : { bgVar: hex, fgVar: "#FFFFFF" }; // fallback to raw hex if unknown
}
```

Update `frontend/components/MemberDot.tsx` to consume the token vars via the helper:

```tsx
import { getMemberColorVars } from "@/lib/colors";

export function MemberDot({ colorHex, ...rest }: MemberDotProps) {
  const { bgVar, fgVar } = getMemberColorVars(colorHex);
  return (
    <span style={{ background: bgVar, color: fgVar }} className="..." {...rest} />
  );
}
```
