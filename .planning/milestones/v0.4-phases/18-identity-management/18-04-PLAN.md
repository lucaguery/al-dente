    import { test, expect } from '@playwright/test';
    import {
      SEEDED_MEMBER_LUCA,
      SEED_AUTH_TOKEN,
      getApiBase,
    } from './fixtures/seed-helpers';

    // IDM-02 — Settings inline rename happy path.
    //
    // Single-browser variant per D-18-18 (cross-phone realtime test deferred
    // to v2). The Playwright assertion is: tap the Pencil affordance, type
    // a new name, press Enter, expect the displayed name to update + toast.
    //
    // Per Phase 18-CONTEXT §D-18-08 the i18n keys live under settings.member.*
    // (frontend/lib/i18n/fr.json). The aria-label strings are the pin (verbatim
    // copy as of Plan 18-02 commit). If those keys move, this spec must update
    // in the same commit (CLAUDE.md locked-vocabulary drift rule).
    const RENAME_PENCIL_ARIA = 'Modifier mon prénom';
    const RENAME_SUCCESS_TOAST = 'Nom mis à jour';

    // Random per-run suffix so reruns within the same DB don't trip the 409
    // uniqueness check (the seeded household has 2 members; the partner name
    // is "Partner" — choose a suffix that cannot collide).
    const RENAMED_NAME = `Luca-test-${Date.now()}`;

    test.describe('settings — IDM-02 member rename', () => {
      test('rename via inline edit reflects on the Settings surface', async ({
        page,
      }) => {
        await page.goto('/settings');

        // Sanity: starting state is the seeded fixture.
        await expect(page.getByText(SEEDED_MEMBER_LUCA, { exact: true })).toBeVisible();

        // Open the rename Input.
        const pencil = page.getByRole('button', { name: RENAME_PENCIL_ARIA });
        await expect(pencil).toBeVisible();
        await pencil.click();

        // The Input is now focused and pre-filled with the current name.
        // Plan 18-02 Task 2 sets aria-label = t("member.rename_label") = "Ton prénom".
        const input = page.getByRole('textbox', { name: 'Ton prénom' });
        await expect(input).toBeVisible();
        await expect(input).toBeFocused();

        // Clear + fill + Enter. Use .fill() (not .pressSequentially()) so the
        // value replaces atomically — autoFocus pre-selects on some browsers
        // but Playwright's .fill() handles both selected and non-selected
        // states by clearing first.
        await input.fill(RENAMED_NAME);
        await input.press('Enter');

        // Toast appears (Sonner) and the new name is visible after refresh().
        await expect(page.getByText(RENAME_SUCCESS_TOAST)).toBeVisible({ timeout: 5000 });
        await expect(page.getByText(RENAMED_NAME, { exact: true })).toBeVisible({
          timeout: 5000,
        });
      });

      // Teardown: restore the seeded name via a direct PATCH so the next
      // test in the suite sees the fixture in its expected shape. Done
      // imperatively (not via the UI) to keep this concise and resilient
      // to any test-runner ordering quirks.
      test.afterAll(async ({ request }) => {
        const apiBase = getApiBase();
        await request.patch(`${apiBase}/households/me`, {
          headers: { Authorization: `Bearer ${SEED_AUTH_TOKEN}` },
          data: { name: SEEDED_MEMBER_LUCA },
        });
      });
    });
    ```

    - The `seeded` project is the default for new specs unless the filename matches the fresh-project testMatch — `settings-member-rename.spec.ts` follows the same naming convention as the existing `settings.spec.ts` (seeded).
    - Per `frontend/AGENTS.md`: Next.js 16 has breaking changes; this spec uses Playwright APIs only (no Next-specific helpers), so the Next.js delta does not apply here.
    - The teardown uses `request` fixture's Bearer-token shortcut — `SEED_AUTH_TOKEN` is the seeded backend token, ungated by cookie path.

    import { test, expect, type BrowserContext, type Page } from '@playwright/test';

    // IDM-04 — onboarding-household-full. Runs in the `fresh` Playwright
    // project (testMatch in playwright.config.ts truncates the DB before
    // the spec runs, so we can assume the household table starts empty).
    //
    // 6 browser contexts:
    //   1. Alice — creates the household, captures invite_code.
    //   2..5. Bob/Carla/Dan/Eve — fill the household to capacity (5 members).
    //   6. Fran — attempts to join, expects the "Foyer complet" terminal Card.
    //
    // The capacity gate is enforced at the BACKEND (Plan 18-01); the frontend
    // (Plan 18-03) discriminates on body.detail.code === "HOUSEHOLD_FULL".

    const MEMBER_COLORS = [
      '#F43F5E', // rose-500 — Alice
      '#F59E0B', // amber-500 — Bob
      '#10B981', // emerald-500 — Carla
      '#0EA5E9', // sky-500 — Dan
      '#8B5CF6', // violet-500 — Eve
    ] as const;

    const HOUSEHOLD_NAME = `Foyer ${Date.now()}`;

    // Pinned i18n strings (mirror of frontend/lib/i18n/fr.json onboarding.join.capacity.*
    // and onboarding.* — added in Plan 18-03 Task 1).
    const CAPACITY_TITLE = 'Foyer complet';
    const CAPACITY_BACK_CTA = 'Revenir à l\'accueil';

    async function createOrJoin(
      page: Page,
      params: {
        action: 'create' | 'join';
        household_name?: string;
        invite_code?: string;
        member_name: string;
        color_hex: string;
      },
    ): Promise<void> {
      if (params.action === 'create') {
        await page.goto('/onboarding/create');
        await page.getByLabel('Nom du foyer').fill(params.household_name!);
        await page.getByLabel('Ton prénom').fill(params.member_name);
        await page.getByRole('radio', { name: hexToColorName(params.color_hex) }).click();
        await page.getByRole('button', { name: 'Créer le foyer' }).click();
      } else {
        await page.goto('/onboarding/join');
        await page.getByLabel('Code d\'invitation').fill(params.invite_code!);
        await page.getByLabel('Ton prénom').fill(params.member_name);
        await page.getByRole('radio', { name: hexToColorName(params.color_hex) }).click();
        await page.getByRole('button', { name: 'Rejoindre', exact: true }).click();
      }
    }

    // ColorSwatchPicker uses aria-labels like "rose", "amber", etc. — match the
    // existing invite-code-happy-path.spec.ts approach.
    function hexToColorName(hex: string): RegExp {
      const map: Record<string, string> = {
        '#F43F5E': 'rose',
        '#F59E0B': 'amber',
        '#10B981': 'emerald',
        '#0EA5E9': 'sky',
        '#8B5CF6': 'violet',
      };
      const name = map[hex.toUpperCase()];
      if (!name) throw new Error(`unmapped color hex: ${hex}`);
      // Case-insensitive regex; the actual aria-label may be the French label
      // for the color (e.g. "Rose"). If invite-code-happy-path.spec.ts uses a
      // different selector style, mirror IT (the goal is parity with the
      // existing onboarding test convention, NOT a re-derivation).
      return new RegExp(name, 'i');
    }

    test.describe('onboarding — IDM-04 household-full terminal Card', () => {
      test('6th joiner sees "Foyer complet" terminal Card', async ({ browser }) => {
        // ---- 1. Alice creates the household (1/5 members) ----
        const aliceCtx: BrowserContext = await browser.newContext();
        const alicePage: Page = await aliceCtx.newPage();
        await createOrJoin(alicePage, {
          action: 'create',
          household_name: HOUSEHOLD_NAME,
          member_name: 'Alice',
          color_hex: MEMBER_COLORS[0],
        });
        // Share-code page should be visible OR we landed on home — either way
        // the invite_code is now in the backend. Read it via /api/households/me
        // (Alice's cookie is set on this context).
        const sessionRes = await alicePage.request.get('/api/households/me');
        const session = await sessionRes.json();
        const inviteCode: string = session.invite_code;
        expect(inviteCode).toMatch(/^[A-Z0-9]{6}$/);

        // ---- 2-5. Bob, Carla, Dan, Eve join (2..5/5) ----
        const fillNames = ['Bob', 'Carla', 'Dan', 'Eve'];
        for (let i = 0; i < fillNames.length; i++) {
          const ctx: BrowserContext = await browser.newContext();
          const p: Page = await ctx.newPage();
          await createOrJoin(p, {
            action: 'join',
            invite_code: inviteCode,
            member_name: fillNames[i],
            color_hex: MEMBER_COLORS[i + 1],
          });
          // Land on home OR share-code OR /; we don't assert a specific URL
          // — just that the join didn't throw a top-level error.
          await ctx.close();
        }

        // ---- 6. Fran attempts to join, expects "Foyer complet" terminal Card ----
        const franCtx: BrowserContext = await browser.newContext();
        const franPage: Page = await franCtx.newPage();
        await franPage.goto('/onboarding/join');
        await franPage.getByLabel('Code d\'invitation').fill(inviteCode);
        await franPage.getByLabel('Ton prénom').fill('Fran');
        // All 5 colors are taken so the radios are disabled — pick any (the
        // backend's capacity check fires BEFORE the color uniqueness check
        // per Plan 18-01 Task 2, so even a taken color triggers 422 HOUSEHOLD_FULL
        // not 409 color-taken).
        // If the radios are physically disabled, force-click by submitting the
        // form via the raw API instead. Test the UI path by clicking the
        // (disabled) submit button after filling code + name; if the button is
        // disabled we skip to the API-driven path below.
        await franPage.getByRole('radio', { name: hexToColorName(MEMBER_COLORS[0]) }).click({ force: true });
        await franPage.getByRole('button', { name: 'Rejoindre', exact: true }).click({ force: true });

        // Terminal Card assertions.
        await expect(
          franPage.getByRole('heading', { name: CAPACITY_TITLE }),
        ).toBeVisible({ timeout: 5000 });
        await expect(franPage.getByText(/5 membres/)).toBeVisible();
        await expect(
          franPage.getByRole('button', { name: CAPACITY_BACK_CTA }),
        ).toBeVisible();

        // Tapping back CTA navigates away from /onboarding/join.
        await franPage.getByRole('button', { name: CAPACITY_BACK_CTA }).click();
        await expect(franPage).not.toHaveURL(/\/onboarding\/join/);

        await franCtx.close();
        await aliceCtx.close();
      });
    });
    ```
