    ```python
    @router.patch("/me", response_model=MemberPublic)
    async def rename_me(
        body: MemberRenameRequest,
        member: Member = Depends(current_member),
        db: Session = Depends(get_db),
    ) -> MemberPublic:
        """Rename the requesting member — IDM-01.

        D-18-01..D-18-04:
        - Body validated 1..40 + stripped at the Pydantic boundary.
        - Uniqueness check is household-scoped and excludes the requester
          (so renaming to your own current name is a no-op 200, not 409).
        - Broadcasts `member.updated` via realtime AFTER the commit per
          CLAUDE.md invariant #4 — partner's RealtimeProvider re-renders.
        - T-04-01-03: scope is auth-derived (member.household_id, member.id)
          — body cannot select a different member.
        """
        new_name = body.name  # Already stripped by Pydantic ConfigDict.

        # Uniqueness within household, excluding self.
        clash = db.scalar(
            select(Member.id).where(
                Member.household_id == member.household_id,
                Member.name == new_name,
                Member.id != member.id,
            )
        )
        if clash is not None:
            raise HTTPException(
                status_code=status.HTTP_409_CONFLICT,
                detail="name already taken",
            )

        member.name = new_name
        db.add(member)
        db.commit()
        db.refresh(member)

        # Invariant #4: broadcast AFTER commit so consumers re-fetch fresh state.
        await broadcast_to_household(
            member.household_id,
            "member.updated",
            {
                "id": str(member.id),
                "name": member.name,
                "color_hex": member.color_hex,
            },
        )
        return member
    ```
