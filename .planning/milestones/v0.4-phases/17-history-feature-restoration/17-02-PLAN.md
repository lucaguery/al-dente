       function ratingChipClass(rating: LogRating): string {
         const base =

         switch (rating) {

         return new Intl.DateTimeFormat("fr-FR", {
           weekday: "long",
           day: "numeric",
           month: "long",
           year: "numeric",
         }).format(date);
       }

       export default function CookingLogDetailPage() {
         const params = useParams<{ id: string }>();
         const id = params?.id;
