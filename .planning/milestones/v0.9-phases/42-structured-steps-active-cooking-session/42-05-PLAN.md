<objective>
Land the Playwright E2E spec that verifies the /active page user journey end-to-end. Three scenarios:

1. **Happy path (structured steps present)** — recipe has populated `StepEntry[]`. Navigate to /active, advance through steps, confirm final-step CTA + navigation to /finalize.
2. **Backfill loading state** — recipe has `steps = []`. Navigate to /active, confirm the loader appears with the documented copy, then (after the BackgroundTask completes in test mode via the canned LLM fixture) confirm the loader is replaced by step 1.
3. **Finalize transition** — from the last step, tap `Terminé · marquer cuisinée`, assert URL change to `/cooking-logs/{id}/finalize`.

The spec relies on the local synthetic seed (`uv run seed` for the standard hermetic data) — same posture as the existing /cooking-logs/[id] specs. Per `frontend/CLAUDE.md`: the synthetic seed does NOT upload photo bytes; the active session does not depend on photo presence so this is fine.

Per `TESTING.md` Pitfall 10: the spec does NOT assert on `consoleErrors` — expected third-party noise (Web Speech API stubs, animation warnings) would flake the assertion.

The backfill scenario depends on plan 42-03's deterministic test-mode short-circuit in `services/llm.py` (`settings.environment == "test"` returns canned data via `llm_fixtures.canned_thread_extract`). The fixture is updated in plan 42-02 Task 5 to emit `StepEntry`-shaped steps. The BackgroundTask + broadcast still runs; from the Playwright spec's perspective, this means the WebSocket `recipe.updated` frame lands within a couple seconds and the loader → navigator transition is observable.
</objective>

  import { test, expect, type Page } from "@playwright/test";
  // Mirror the existing import pattern — e.g.:

  // Helper: seed-data assumption — the synthetic seed creates a cooking_log
  // row for at least one recipe per household. Per Task 1, the closest deterministic
  // anchor is recipe titled "<from seed>" with cooking_log id discoverable via
  // GET /api/cooking-logs/active (returns the in-progress log for today, if any).
  // If the seed does not create an in-progress log, the spec MUST start one via
  // POST /api/recipes/{id}/cook before navigating to /active.

  async function startCookingSessionForFirstRecipe(page: Page): Promise<string> {
    // Hit the API directly via page.request to start a session, return the new cooking_log id.
    const recipes = await page.request.get("/api/recipes").then(r => r.json());
    const target = recipes.find((r: any) => r.status === "structured" && Array.isArray(r.steps) && r.steps.length > 0)
      ?? recipes.find((r: any) => r.status === "structured");
    if (!target) throw new Error("No structured recipe in seed");
    const cookResp = await page.request.post(`/api/recipes/${target.id}/cook`);
    const log = await cookResp.json();
    return log.id;
  }

  test.describe("Active cooking session — /cooking-logs/[id]/active", () => {

    test("happy path: structured steps render, navigator advances, finalize CTA appears on last step", async ({ page }) => {
      const cookingLogId = await startCookingSessionForFirstRecipe(page);
      await page.goto(`/cooking-logs/${cookingLogId}/active`);

      // det-top crumb
      await expect(page.getByText(/démarrée à/)).toBeVisible({ timeout: 10_000 });

      // étape pin (e.g. "étape 1/N")
      await expect(page.getByText(/étape\s+\d+\/\d+/)).toBeVisible();

      // step text body — Geist 400 prose paragraph
      const stepBody = page.locator("main p").first();
      await expect(stepBody).toBeVisible();

      // Step-next button (next-intl key cooking_active.step_next → "Étape suivante")
      const nextBtn = page.getByRole("button", { name: "Étape suivante" });
      await expect(nextBtn).toBeVisible();

      // Get total step count from the pin to know how many advances are needed
      const pinText = await page.getByText(/étape\s+\d+\/\d+/).innerText();
      const match = pinText.match(/étape\s+(\d+)\/(\d+)/);
      expect(match).not.toBeNull();
      const total = parseInt(match![2], 10);

      // Advance through all but the last step
      for (let i = 0; i < total - 1; i++) {
        await page.getByRole("button", { name: "Étape suivante" }).click();
      }

      // Final step — CTA flips to finalize
      await expect(page.getByRole("button", { name: /Terminé.*marquer cuisinée/ })).toBeVisible();
      await expect(page.getByRole("button", { name: "Étape suivante" })).toHaveCount(0);
    });

    test("backfill loading: recipe with empty steps shows loader, then steps appear via recipe.updated", async ({ page }) => {
      // Seed assumption: at least one structured recipe exists with steps = [].

      test.fixme(true, "Requires synthetic seed update — add an explicit empty-steps recipe row. " +
        "Phase 42 ships the production path; the seed update is tracked as # TODO(productize) " +
        "in backend/scripts/seed.py (or wherever the seed lives).");

      // The full assertion shape, gated behind the fixme until seed support lands:
      // const cookingLogId = await startCookingSessionForEmptyStepsRecipe(page);
      // await page.goto(`/cooking-logs/${cookingLogId}/active`);

      // await expect(page.getByText(/étape\s+1\/\d+/)).toBeVisible();
    });

    test("finalize transition: tapping Terminé routes to /finalize", async ({ page }) => {
      const cookingLogId = await startCookingSessionForFirstRecipe(page);
      await page.goto(`/cooking-logs/${cookingLogId}/active`);

      // Advance to the last step (reuse the count-and-loop pattern from happy path)
      await expect(page.getByText(/étape\s+\d+\/\d+/)).toBeVisible({ timeout: 10_000 });

        await page.getByRole("button", { name: "Étape suivante" }).click();

      await page.getByRole("button", { name: /Terminé.*marquer cuisinée/ }).click();

      // URL transitions to /finalize
      await expect(page).toHaveURL(new RegExp(`/cooking-logs/${cookingLogId}/finalize`));
    });

  });
