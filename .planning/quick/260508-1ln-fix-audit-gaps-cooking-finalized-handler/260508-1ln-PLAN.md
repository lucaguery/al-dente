    ```typescript
    // Phase 4 — cooking.finalized (partner finalized the cooking session).
    useEffect(() => {
      if (!client) return;
      const off = client.onEvent<Phase4CookingFinalizedEvent>("cooking.finalized", (payload) => {
        if (typeof window === "undefined") return;
        window.dispatchEvent(

        );
      });
      return off;
    }, [client]);
    ```
